\section*{Chapter 1}

\subsection*{Sigma Algebras}
\begin{definition}
Let $X$ be an arbitrary set. A collection $\mathcal{A}$ of subsets of $X$ is called a \emph{$\sigma$-algebra} on $X$ if:
\begin{enumerate}[label=(\alph*)]
    \item $X \in \mathcal{A}$,
    \item if $A \in \mathcal{A}$, then $A^c \in \mathcal{A}$,
    \item if $\{A_i\}_{i=1}^\infty \subset \mathcal{A}$, then $\bigcup_{i=1}^\infty A_i \in \mathcal{A}$,
    \item if $\{A_i\}_{i=1}^\infty \subset \mathcal{A}$, then $\bigcap_{i=1}^\infty A_i \in \mathcal{A}$.
\end{enumerate}
\end{definition}

\begin{proof}
Assume (a), (b), and (c). Let $\{A_i\}_{i=1}^\infty \subset \mathcal{A}$. Then $A_i^c \in \mathcal{A}$ for all $i$, and by (c),
\(\bigcup_{i=1}^\infty A_i^c \in \mathcal{A}\). By (b) again, \(\left( \bigcup_{i=1}^\infty A_i^c \right)^c \in \mathcal{A}.\)
By De Morgan’s law,
\[
\bigcap_{i=1}^\infty A_i = \left( \bigcup_{i=1}^\infty A_i^c \right)^c \in \mathcal{A}
\]

\end{proof}

\begin{proposition}
Let $X$ be a set. Then the intersection of an arbitrary nonempty collection of
$\sigma$-algebras on $X$ is a $\sigma$-algebra on $X$.
\end{proposition}

\begin{proof}
Let $\mathcal{C}$ be a nonempty collection of $\sigma$-algebras on $X$, and define
\[
\mathcal{A} := \bigcap_{\mathcal{B} \in \mathcal{C}} \mathcal{B}.
\]
Since every $\mathcal{B} \in \mathcal{C}$ contains $X$, we have $X \in \mathcal{A}$.
If $A \in \mathcal{A}$, then $A \in \mathcal{B}$ for every
$\mathcal{B} \in \mathcal{C}$. Since each $\mathcal{B}$ is a $\sigma$-algebra,
$A^c \in \mathcal{B}$ for every $\mathcal{B} \in \mathcal{C}$, hence
$A^c \in \mathcal{A}$. If $\{A_i\}_{i=1}^{\infty} \subset \mathcal{A}$, then
$\{A_i\}_{i=1}^{\infty} \subset \mathcal{B}$ for every
$\mathcal{B} \in \mathcal{C}$. Hence
\[
\bigcup_{i=1}^{\infty} A_i \in \mathcal{B}
\]
for every $\mathcal{B} \in \mathcal{C}$, and therefore
\[
\bigcup_{i=1}^{\infty} A_i \in \mathcal{A}.
\]
Thus $\mathcal{A}$ is a $\sigma$-algebra on $X$.
\end{proof}

\begin{corollary}
Let $X$ be a set, and let $\mathcal{F}$ be a family of subsets of $X$.
Then there exists a smallest $\sigma$-algebra on $X$ that contains $\mathcal{F}$.
\end{corollary}

\begin{proof}
Let $\mathcal{C}$ be the collection of all $\sigma$-algebras on $X$ that contain
$\mathcal{F}$. This collection is nonempty, since $\mathcal{P}(X)$ is a
$\sigma$-algebra on $X$ and contains $\mathcal{F}$.

Define
\[
\sigma(\mathcal{F}) := \bigcap_{\mathcal{A} \in \mathcal{C}} \mathcal{A}.
\]
By the previous proposition, $\sigma(\mathcal{F})$ is a $\sigma$-algebra on $X$.
It contains $\mathcal{F}$, since every $\mathcal{A} \in \mathcal{C}$ contains
$\mathcal{F}$. Moreover, if $\mathcal{B}$ is any $\sigma$-algebra on $X$ that
contains $\mathcal{F}$, then $\mathcal{B} \in \mathcal{C}$, and therefore
\[
\sigma(\mathcal{F}) \subset \mathcal{B}.
\]
Thus $\sigma(\mathcal{F})$ is the smallest $\sigma$-algebra on $X$ containing
$\mathcal{F}$.
\end{proof}

\begin{definition}
The \emph{Borel $\sigma$-algebra} on $\mathbb{R}^d$ is the $\sigma$-algebra
generated by the open subsets of $\mathbb{R}^d$. It is denoted by
$\mathcal{B}(\mathbb{R}^d)$. Its elements are called \emph{Borel sets}.
\end{definition}

\begin{proposition}
The Borel $\sigma$-algebra $\mathcal{B}(\mathbb{R})$ is generated by each of the
following collections:
\begin{enumerate}[label=(\alph*)]
    \item the collection of all closed subsets of $\mathbb{R}$,
    \item the collection of all intervals of the form $(-\infty,b]$,
    \item the collection of all intervals of the form $(a,b]$.
\end{enumerate}
\end{proposition}

\begin{proof}
Let $\mathcal{B}_1,\mathcal{B}_2,\mathcal{B}_3$ denote the $\sigma$-algebras
generated by the collections in (a), (b), and (c), respectively.

Since $\mathcal{B}(\mathbb{R})$ contains all open sets and is closed under
complements, it contains all closed sets. Hence
\[
\mathcal{B}(\mathbb{R}) \supset \mathcal{B}_1.
\]
Every set $(-\infty,b]$ is closed, so $\mathcal{B}_1 \supset \mathcal{B}_2$.
Moreover,
\[
(a,b] = (-\infty,b] \cap (-\infty,a]^c,
\]
so $\mathcal{B}_2 \supset \mathcal{B}_3$.

Finally, let $G \subset \mathbb{R}$ be open. For each $x \in G$, since $G$ is
open, there exist rational numbers $a,b \in \mathbb{Q}$ such that
\[
x \in (a,b] \subset G.
\]
Hence
\[
G = \bigcup_{\substack{a,b \in \mathbb{Q} \\ (a,b] \subset G}} (a,b].
\]
This is a countable union, since $\mathbb{Q}^2$ is countable. Therefore
$G \in \mathcal{B}_3$. Thus every open set belongs to $\mathcal{B}_3$, so
\[
\mathcal{B}(\mathbb{R}) \subset \mathcal{B}_3.
\]
Therefore all four $\sigma$-algebras are equal.
\end{proof}

\subsection*{Measures}

\begin{definition}
Let $X$ be a set and $\mathcal{A}$ a $\sigma$-algebra on $X$. A function
$\mu : \mathcal{A} \to [0,\infty]$ is called a \emph{measure} if
\begin{enumerate}[label=(\alph*)]
    \item $\mu(\emptyset) = 0$,
    \item for every $\{A_i\}_{i=1}^\infty$ of pairwise disjoint sets in $\mathcal{A}$,
    \(
    \mu\!\left(\bigcup_{i=1}^\infty A_i\right) = \sum_{i=1}^\infty \mu(A_i).
    \)
\end{enumerate}
\end{definition}

\begin{proposition}[Proposition 1.2.5]
Let $(X,\mathcal{A},\mu)$ be a measure space.
\begin{enumerate}[label=(\alph*)]
    \item If $\{A_k\}$ is an increasing sequence in $\mathcal{A}$, then
    \[
    \mu\!\left(\bigcup_{k=1}^\infty A_k\right) = \lim_{k \to \infty} \mu(A_k).
    \]
    \item If $\{A_k\}$ is a decreasing sequence in $\mathcal{A}$ and
    $\mu(A_n) < \infty$ for some $n$, then
    \[
    \mu\!\left(\bigcap_{k=1}^\infty A_k\right) = \lim_{k \to \infty} \mu(A_k).
    \]
\end{enumerate}
\end{proposition}

\begin{proof}
(a) Define $B_1 = A_1$ and $B_k = A_k \setminus A_{k-1}$ for $k \geq 2$.
Then $\{B_k\}$ are disjoint and $\bigcup_k A_k = \bigcup_k B_k$. Hence
\[
\mu\!\left(\bigcup_k A_k\right) = \sum_k \mu(B_k)
= \lim_{k \to \infty} \sum_{i=1}^k \mu(B_i)
= \lim_{k \to \infty} \mu(A_k).
\]

(b) Since $\mu(A_n) < \infty$ for some $n$, we may replace $\{A_k\}$ by the
tail $\{A_{k+n-1}\}$ and assume $\mu(A_1) < \infty$. Let $C_k = A_1 \setminus A_k$.
Then $\{C_k\}$ is increasing and
\[
\bigcup_k C_k = A_1 \setminus \bigcap_k A_k.
\]
By (a),
\[
\mu(A_1 \setminus \bigcap_k A_k) = \lim_{k \to \infty} \mu(C_k)
= \lim_{k \to \infty} \big(\mu(A_1) - \mu(A_k)\big),
\]
which gives $\mu(\bigcap_k A_k) = \lim_k \mu(A_k)$.
\end{proof}

\subsection*{Outer Measures}
\begin{definition}
Let $X$ be a set. An \emph{outer measure} on $X$ is a function
$\mu^* : \mathcal{P}(X) \to [0,\infty]$ such that
\begin{enumerate}[label=(\alph*)]
    \item $\mu^*(\emptyset) = 0$,
    \item if $A \subset B$, then $\mu^*(A) \le \mu^*(B)$,
    \item for any sequence $\{A_n\}$ of subsets of $X$,
    \(
    \mu^*\!\left(\bigcup_{n=1}^\infty A_n\right)
    \le \sum_{n=1}^\infty \mu^*(A_n).
    \)
\end{enumerate}
\end{definition}

\begin{definition}
The \emph{Lebesgue outer measure} on $\mathbb{R}$ is the function
$\lambda^* : \mathcal{P}(\mathbb{R}) \to [0,\infty]$ defined by
\[
\lambda^*(A)
= \inf \left\{
\sum_{i=1}^\infty (b_i - a_i)
:\;
A \subset \bigcup_{i=1}^\infty (a_i,b_i)
\right\}.
\]
\end{definition}

\begin{definition}
The \emph{Lebesgue outer measure} on $\mathbb{R}^d$ is defined by
\[
\lambda^*(A)
= \inf \left\{
\sum_{i=1}^\infty \mathrm{vol}(R_i)
:\;
A \subset \bigcup_{i=1}^\infty R_i
\right\},
\]
where each $R_i$ is a bounded open rectangle in $\mathbb{R}^d$ and
\[
\mathrm{vol}(I_1 \times \cdots \times I_d)
= \prod_{j=1}^d |I_j|.
\]
\end{definition}

\begin{definition}
Let $\mu^*$ be an outer measure on $X$. A set $B \subset X$ is called
\emph{$\mu^*$-measurable} if for all $A \subset X$,
\[
\mu^*(A) = \mu^*(A \cap B) + \mu^*(A \cap B^c).
\]
\end{definition}

\begin{remark}
Outer measures are defined on all subsets of $X$, but are only
subadditive in general. Carathéodory’s criterion selects a class of
sets on which $\mu^*$ becomes additive. This allows us to restrict
$\mu^*$ to a $\sigma$-algebra and obtain a genuine measure (in
particular, the Lebesgue measure).
\end{remark}

\begin{theorem}
Let $X$ be a set and $\mu^*$ an outer measure on $X$. Let
\[
\mathcal{M}_{\mu^*} := \{ B \subset X : B \text{ is } \mu^*\text{-measurable} \}.
\]
Then:
\begin{enumerate}[label=(\alph*)]
    \item $\mathcal{M}_{\mu^*}$ is a $\sigma$-algebra,
    \item the restriction of $\mu^*$ to $\mathcal{M}_{\mu^*}$ is a measure.
\end{enumerate}
\end{theorem}

\begin{proof}[Proof sketch]
One shows that $\mathcal{M}_{\mu^*}$ is closed under complements directly from
the definition. Closure under finite unions follows by applying the defining
identity to $B_1$ and $B_2$ and decomposing sets appropriately.

Closure under countable unions is obtained by first proving the result for
finite disjoint unions and then passing to the limit using subadditivity of
$\mu^*$.

Finally, countable additivity of $\mu^*$ on $\mathcal{M}_{\mu^*}$ follows by
applying the defining property to disjoint measurable sets and using the
previous decomposition.
\end{proof}

\begin{remark}
Applying this theorem to the Lebesgue outer measure $\lambda^*$ on $\mathbb{R}^d$
produces a $\sigma$-algebra $\mathcal{M}_{\lambda^*}$, called the
\emph{Lebesgue $\sigma$-algebra}. The restriction of $\lambda^*$ to this
$\sigma$-algebra is the Lebesgue measure. Thus, outer measure together with
Carathéodory’s criterion provides a systematic way to construct Lebesgue measure.
\end{remark}

\begin{example}[Cantor set]
Let $K_0 = [0,1]$. Construct $K_n$ by removing the open middle third of each
interval in $K_{n-1}$. The \emph{Cantor set} is
\(
K = \bigcap_{n=0}^\infty K_n.
\)
\end{example}

\begin{proposition}
The Cantor set $K$ is compact, uncountable, and has Lebesgue measure zero.
\end{proposition}

\begin{proof}[Idea]
Each $K_n$ is a finite union of closed intervals, hence compact, and
$K = \bigcap_n K_n$ is compact. One shows $K$ has the cardinality of the
continuum via ternary expansions.

Moreover, $\lambda(K_n) = (2/3)^n$, so
\[
\lambda(K) \le \lambda(K_n) = (2/3)^n \to 0,
\]
hence $\lambda(K)=0$.
\end{proof}

\begin{theorem}
There exists a subset of $(0,1)$ that is not Lebesgue measurable.
\end{theorem}

\begin{proof}[Idea]
Define $x \sim y$ if $x-y \in \mathbb{Q}$ and choose one representative from
each equivalence class (using the axiom of choice) to form a set $E \subset (0,1)$.

Consider the translates $E + r_n$ by rational numbers $r_n$. These sets are
disjoint and their union covers $(0,1)$ up to inclusion. Translation invariance
and countable additivity would force a contradiction, so $E$ cannot be measurable.
\end{proof}

\begin{remark}
Not every subset of $\mathbb{R}$ can be assigned a consistent measure: the
existence of non-measurable sets shows that countable additivity and translation
invariance cannot hold on all subsets. 

Therefore, one must restrict attention to a $\sigma$-algebra (such as the
Lebesgue $\sigma$-algebra) on which a well-defined measure exists.
\end{remark}

\section*{Chapter 2}
\begin{definition}
Let $(X,\mathcal{A},\mu)$ be a measure space. A nonnegative simple measurable
function is a function of the form
\[
f = \sum_{i=1}^n a_i \mathbf{1}_{A_i},
\]
where $a_i \ge 0$ and $A_i \in \mathcal{A}$ are disjoint. Its integral is defined by
\[
\int f\,d\mu := \sum_{i=1}^n a_i\mu(A_i).
\]
\end{definition}

\begin{definition}
Let $f : X \to [0,\infty]$ be measurable. The integral of $f$ is defined by
\[
\int f\,d\mu
:=
\sup\left\{
\int g\,d\mu : g \in \mathcal{S}_+,\ g \le f
\right\},
\]
where $\mathcal{S}_+$ denotes the set of nonnegative simple measurable functions.
\end{definition}

\begin{proposition}
Every nonnegative measurable function $f : X \to [0,\infty]$ is the pointwise
limit of an increasing sequence of nonnegative simple measurable functions.
\end{proposition}

\begin{proof}[Reference]
This is Proposition 2.1.8 in Cohn. It guarantees that nonnegative measurable
functions can be approximated from below by simple functions.
\end{proof}

\begin{proposition}
If $f_n \in \mathcal{S}_+$, $f_n \uparrow f$, and $f$ is nonnegative measurable,
then
\[
\int f\,d\mu = \lim_{n\to\infty} \int f_n\,d\mu.
\]
\end{proposition}

\begin{proof}[Reference]
This is Proposition 2.3.3 in Cohn. It shows that the supremum definition agrees
with taking limits along increasing simple approximations. Proposition 2.3.2 is
the corresponding result when the limiting function is itself simple.
\end{proof}

\begin{remark}
Thus the integral is first defined directly for nonnegative simple functions.
For arbitrary nonnegative measurable functions, it is defined as the supremum of
the integrals of all simple functions below it. Propositions 2.1.8 and 2.3.3
show that this construction is compatible with approximation from below.
\end{remark}

\begin{definition}
Let $f : X \to [-\infty,\infty]$ be measurable. Define
\[
f^+ := \max(f,0), \qquad f^- := \max(-f,0).
\]
Then
\[
f = f^+ - f^-,
\qquad
|f| = f^+ + f^-.
\]
If at least one of $\int f^+\,d\mu$ and $\int f^-\,d\mu$ is finite, then
$\int f\,d\mu$ is said to \emph{exist}, and is defined by
\[
\int f\,d\mu
:=
\int f^+\,d\mu - \int f^-\,d\mu.
\]
If both $\int f^+\,d\mu$ and $\int f^-\,d\mu$ are finite, then $f$ is called
\emph{integrable}.
\end{definition}

\begin{definition}
If $A \in \mathcal{A}$, then $f$ is called \emph{integrable over $A$} if
$f\mathbf{1}_A$ is integrable. In that case,
\[
\int_A f\,d\mu := \int f\mathbf{1}_A\,d\mu.
\]
\end{definition}

\begin{proposition}[Proposition 2.3.2]
Let $(X,\mathcal{A},\mu)$ be a measure space. Let $f\in\mathcal{S}_+$, and let
$\{f_n\}$ be an increasing sequence in $\mathcal{S}_+$ such that
\[
f = \lim_{n\to\infty} f_n
\]
pointwise on $X$. Then
\[
\int f\,d\mu = \lim_{n\to\infty} \int f_n\,d\mu.
\]
\end{proposition}

\begin{proof}
By monotonicity,
\[
\int f_1\,d\mu \le \int f_2\,d\mu \le \cdots \le \int f\,d\mu,
\]
so
\[
\lim_{n\to\infty}\int f_n\,d\mu \le \int f\,d\mu.
\]

We prove the reverse inequality. Let $0<\varepsilon<1$. Write
\[
f=\sum_{i=1}^k a_i\mathbf{1}_{A_i},
\]
where $a_i>0$ and the sets $A_i\in\mathcal{A}$ are disjoint. For each $n$ and
$i$, define
\[
A(n,i):=\{x\in A_i : f_n(x)\ge (1-\varepsilon)a_i\}.
\]
Then $A(n,i)\in\mathcal{A}$, the sequence $\{A(n,i)\}_{n=1}^\infty$ is increasing,
and
\[
A_i=\bigcup_{n=1}^\infty A(n,i).
\]
Define
\[
g_n:=\sum_{i=1}^k (1-\varepsilon)a_i\mathbf{1}_{A(n,i)}.
\]
Then $g_n\in\mathcal{S}_+$ and $g_n\le f_n$. Hence
\[
\int g_n\,d\mu \le \int f_n\,d\mu.
\]
By Proposition 1.2.5,
\[
\lim_{n\to\infty}\int g_n\,d\mu
=
\lim_{n\to\infty}\sum_{i=1}^k (1-\varepsilon)a_i\mu(A(n,i))
=
\sum_{i=1}^k (1-\varepsilon)a_i\mu(A_i)
=
(1-\varepsilon)\int f\,d\mu.
\]
Therefore
\[
(1-\varepsilon)\int f\,d\mu
\le
\lim_{n\to\infty}\int f_n\,d\mu.
\]
Since $\varepsilon>0$ was arbitrary,
\[
\int f\,d\mu
\le
\lim_{n\to\infty}\int f_n\,d\mu.
\]
Thus
\[
\int f\,d\mu = \lim_{n\to\infty}\int f_n\,d\mu.
\]
\end{proof}

\begin{proposition}[Proposition 2.3.3]
Let $(X,\mathcal{A},\mu)$ be a measure space. Let $f:X\to[0,\infty]$ be
measurable, and let $\{f_n\}$ be an increasing sequence of functions in
$\mathcal{S}_+$ such that
\[
f = \lim_{n\to\infty} f_n
\]
pointwise on $X$. Then
\[
\int f\,d\mu = \lim_{n\to\infty} \int f_n\,d\mu.
\]
\end{proposition}

\begin{proof}
By monotonicity of the integral,
\[
\int f_1\,d\mu \le \int f_2\,d\mu \le \cdots \le \int f\,d\mu.
\]
Hence
\[
\lim_{n\to\infty}\int f_n\,d\mu \le \int f\,d\mu.
\]

For the reverse inequality, let $g \in \mathcal{S}_+$ with $g \le f$. Then
$\{g \wedge f_n\}$ is an increasing sequence in $\mathcal{S}_+$ and
\[
g = \lim_{n\to\infty} (g \wedge f_n).
\]
By Proposition 2.3.2,
\[
\int g\,d\mu
=
\lim_{n\to\infty}\int (g \wedge f_n)\,d\mu.
\]
Since $g \wedge f_n \le f_n$, we arrive at
\[
\int f\,d\mu
=
\sup\left\{\int g\,d\mu : g\in\mathcal{S}_+,\ g\le f\right\}\le \lim_{n\to\infty}\int f_n\,d\mu.
\]
\end{proof}

\begin{proposition}[Proposition 2.3.9]
Let $(X,\mathcal{A},\mu)$ be a measure space, and let $f,g:X\to[-\infty,\infty]$
be measurable functions such that $f=g$ almost everywhere. If either
$\int f\,d\mu$ or $\int g\,d\mu$ exists, then both exist and
\[
\int f\,d\mu = \int g\,d\mu.
\]
\end{proposition}

\begin{proof}
First suppose $f,g\ge 0$. Let
\[
A=\{x\in X:f(x)\neq g(x)\}.
\]
Then $\mu(A)=0$. Define
\[
h(x)=
\begin{cases}
+\infty, & x\in A,\\
0, & x\notin A.
\end{cases}
\]

Since $\mu(A)=0$, define
\[
h_n := n\mathbf{1}_A.
\]
Then $h_n \in \mathcal{S}_+$, $h_n \uparrow h$, and
\[
\int h_n\,d\mu = n\mu(A)=0.
\]
By Proposition 2.3.3,
\[
\int h\,d\mu
=
\lim_{n\to\infty}\int h_n\,d\mu
=
0.
\]

Since $f\le g+h$ and $g\le f+h$, monotonicity gives
\[
\int f\,d\mu \le \int g\,d\mu+\int h\,d\mu=\int g\,d\mu
\]
and similarly
\[
\int g\,d\mu \le \int f\,d\mu.
\]
Thus $\int f\,d\mu=\int g\,d\mu$.

For general $f,g$, apply the nonnegative case to $f^+,g^+$ and to $f^-,g^-$.
\end{proof}

\begin{proposition}[Proposition 2.3.10]
Let $(X,\mathcal{A},\mu)$ be a measure space, and let
$f:X\to[0,\infty]$ be measurable. If $t>0$ and
\[
A_t:=\{x\in X:f(x)\ge t\},
\]
then
\[
\mu(A_t)
\le
\frac{1}{t}\int_{A_t} f\,d\mu
\le
\frac{1}{t}\int f\,d\mu.
\]
\end{proposition}

\begin{proof}
On $A_t$ we have $f\ge t$, hence
\[
0\le t\mathbf{1}_{A_t}\le f\mathbf{1}_{A_t}\le f.
\]
By monotonicity of the integral,
\[
\int t\mathbf{1}_{A_t}\,d\mu
\le
\int_{A_t} f\,d\mu
\le
\int f\,d\mu.
\]
Since
\[
\int t\mathbf{1}_{A_t}\,d\mu=t\mu(A_t),
\]
division by $t$ gives
\[
\mu(A_t)
\le
\frac{1}{t}\int_{A_t} f\,d\mu
\le
\frac{1}{t}\int f\,d\mu.
\]
\end{proof}

\begin{theorem}[Monotone Convergence Theorem]
Let $(X,\mathcal{A},\mu)$ be a measure space, and let
$f,f_1,f_2,\dots$ be $[0,\infty]$-valued measurable functions on $X$.
Suppose that
\[
f_1 \le f_2 \le \cdots
\]
and
\[
f = \lim_{n\to\infty} f_n
\]
hold almost everywhere. Then
\[
\int f\,d\mu = \lim_{n\to\infty} \int f_n\,d\mu.
\]
\end{theorem}

\begin{proof}
First assume that $f_n \uparrow f$ pointwise everywhere. By monotonicity of the
integral,
\[
\int f_1\,d\mu \le \int f_2\,d\mu \le \cdots \le \int f\,d\mu.
\]
Hence
\[
\lim_{n\to\infty}\int f_n\,d\mu \le \int f\,d\mu.
\]

We prove the reverse inequality. For each $n$, choose an increasing sequence
$\{g_{n,k}\}_{k=1}^\infty$ of nonnegative simple functions such that
\[
g_{n,k} \uparrow f_n.
\]
This is possible by Proposition 2.1.8. Define
\[
h_n := \max(g_{1,n},g_{2,n},\dots,g_{n,n}).
\]
Then $\{h_n\}$ is an increasing sequence of nonnegative simple functions,
$h_n \le f_n$, and
\[
h_n \uparrow f.
\]
Therefore, by Proposition 2.3.3,
\[
\int f\,d\mu = \lim_{n\to\infty} \int h_n\,d\mu.
\]
Since $h_n \le f_n$, monotonicity gives
\[
\int h_n\,d\mu \le \int f_n\,d\mu.
\]
Thus
\[
\int f\,d\mu
=
\lim_{n\to\infty} \int h_n\,d\mu
\le
\lim_{n\to\infty} \int f_n\,d\mu.
\]
Combining the two inequalities gives
\[
\int f\,d\mu = \lim_{n\to\infty} \int f_n\,d\mu.
\]

If the assumptions hold only almost everywhere, choose a null set $N$ outside
which they hold. Apply the first part to $f_n\mathbf{1}_{N^c}$ and
$f\mathbf{1}_{N^c}$. Since changing a function on a null set does not change its
integral, the result follows.
\end{proof}

\begin{theorem}[Fatou's Lemma]
Let $(X,\mathcal{A},\mu)$ be a measure space, and let $\{f_n\}$ be a sequence of
$[0,\infty]$-valued measurable functions on $X$. Then
\[
\int \liminf_{n\to\infty} f_n \, d\mu
\le
\liminf_{n\to\infty} \int f_n\,d\mu.
\]
\end{theorem}

\begin{proof}
For each $n$, define
\[
g_n := \inf_{k\ge n} f_k.
\]
By Proposition 2.1.5 each $g_n$ is measurable and
\[
g_1 \le g_2 \le \cdots.
\]
Moreover,
\[
\lim_{n\to\infty} g_n
=
\liminf_{n\to\infty} f_n.
\]
By the Monotone Convergence Theorem,
\[
\int \liminf_{n\to\infty} f_n\,d\mu
=
\int \lim_{n\to\infty} g_n\,d\mu
=
\lim_{n\to\infty} \int g_n\,d\mu.
\]
Since $g_n \le f_k$ for every $k\ge n$, we have
\[
\int g_n\,d\mu \le \inf_{k\ge n} \int f_k\,d\mu.
\]
Therefore
\[
\lim_{n\to\infty}\int g_n\,d\mu
\le
\lim_{n\to\infty}\inf_{k\ge n}\int f_k\,d\mu
=
\liminf_{n\to\infty}\int f_n\,d\mu.
\]
Hence
\[
\int \liminf_{n\to\infty} f_n \, d\mu
\le
\liminf_{n\to\infty} \int f_n\,d\mu.
\]
\end{proof}

\begin{theorem}[Dominated Convergence Theorem]
Let $(X,\mathcal{A},\mu)$ be a measure space. Let $g:X\to[0,\infty]$ be
integrable, and let $f,f_1,f_2,\dots$ be $[-\infty,\infty]$-valued measurable
functions on $X$ such that
\[
f_n \to f
\]
almost everywhere, and
\[
|f_n| \le g
\]
almost everywhere for every $n$. Then $f$ and $f_1,f_2,\dots$ are integrable, and
\[
\int f\,d\mu
=
\lim_{n\to\infty}\int f_n\,d\mu.
\]
\end{theorem}

\begin{proof}
First fix $n$. Since $|f_n| \le g$ almost everywhere, Proposition 2.3.9 says that
changing functions on a null set does not change their integrals. Hence we may
treat the inequality as holding everywhere for the purpose of integration.

Since $0 \le |f_n| \le g$, monotonicity of the integral, Proposition 2.3.4(c),
gives
\[
\int |f_n|\,d\mu \le \int g\,d\mu < \infty.
\]
Thus $|f_n|$ is integrable. Proposition 2.3.8 says that a measurable function is
integrable if and only if its absolute value is integrable. Therefore $f_n$ is
integrable.

It remains to prove that $f$ is integrable. Since $f_n \to f$ almost everywhere
and $|f_n| \le g$ almost everywhere for every $n$, we get
\[
|f| \le g
\]
almost everywhere. Again, by Proposition 2.3.9, we may ignore the null set where
this inequality may fail. Since $0 \le |f| \le g$, monotonicity of the integral,
Proposition 2.3.4(c), gives
\[
\int |f|\,d\mu \le \int g\,d\mu < \infty.
\]
Thus $|f|$ is integrable. By Proposition 2.3.8, since a function is integrable
if and only if its absolute value is integrable, $f$ is integrable.

We now prove that
\[
\int f\,d\mu = \lim_{n\to\infty}\int f_n\,d\mu.
\]
First assume that
\[
f_n(x)\to f(x), \qquad |f_n(x)|\le g(x), \qquad g(x)<\infty
\]
hold for every $x\in X$.

Then $g+f_n\ge 0$ and $g+f\ge 0$. Since $g+f_n\to g+f$, Fatou's Lemma gives
\[
\int (g+f)\,d\mu
\le
\liminf_{n\to\infty}\int (g+f_n)\,d\mu.
\]
By linearity and since $\int g\,d\mu<\infty$,
\[
\int g\,d\mu+\int f\,d\mu
\le
\int g\,d\mu+\liminf_{n\to\infty}\int f_n\,d\mu.
\]
Hence
\[
\int f\,d\mu
\le
\liminf_{n\to\infty}\int f_n\,d\mu.
\]

Similarly, $g-f_n\ge 0$ and $g-f\ge 0$. Since $g-f_n\to g-f$, Fatou's Lemma gives
\[
\int (g-f)\,d\mu
\le
\liminf_{n\to\infty}\int (g-f_n)\,d\mu.
\]
Again using linearity,
\[
\int g\,d\mu-\int f\,d\mu
\le
\int g\,d\mu-\limsup_{n\to\infty}\int f_n\,d\mu.
\]
Thus
\[
\limsup_{n\to\infty}\int f_n\,d\mu
\le
\int f\,d\mu.
\]
Combining the two inequalities gives
\[
\limsup_{n\to\infty}\int f_n\,d\mu
\le
\int f\,d\mu
\le
\liminf_{n\to\infty}\int f_n\,d\mu.
\]
Therefore
\[
\int f\,d\mu
=
\lim_{n\to\infty}\int f_n\,d\mu.
\]

Now suppose the assumptions hold only almost everywhere. Let $N\in\mathcal{A}$
be a null set containing all points where one of the following fails:
\[
f_n(x)\to f(x), \qquad |f_n(x)|\le g(x)\text{ for some }n, \qquad g(x)<\infty.
\]
The last condition may be included since $g$ is integrable, so $g<\infty$
almost everywhere.

On $N^c$, all assumptions hold everywhere. Therefore the first part applies to
the functions $f_n\mathbf{1}_{N^c}$, $f\mathbf{1}_{N^c}$, and
$g\mathbf{1}_{N^c}$, giving
\[
\int f\mathbf{1}_{N^c}\,d\mu
=
\lim_{n\to\infty}\int f_n\mathbf{1}_{N^c}\,d\mu.
\]
Since $f\mathbf{1}_{N^c}=f$ almost everywhere and
$f_n\mathbf{1}_{N^c}=f_n$ almost everywhere, Proposition 2.3.9 gives
\[
\int f\,d\mu
=
\lim_{n\to\infty}\int f_n\,d\mu.
\]

\end{proof}

\begin{definition}
Let $f:[a,b]\to\mathbb{R}$ be bounded, and let
\[
\mathcal{P}=\{a=x_0<x_1<\cdots<x_n=b\}
\]
be a partition of $[a,b]$. Define
\[
m_i=\inf_{x\in[x_{i-1},x_i]} f(x),
\qquad
M_i=\sup_{x\in[x_{i-1},x_i]} f(x).
\]
The \emph{lower sum} and \emph{upper sum} of $f$ with respect to $\mathcal{P}$
are
\[
l(f,\mathcal{P})
=
\sum_{i=1}^n m_i(x_i-x_{i-1}),
\]
and
\[
u(f,\mathcal{P})
=
\sum_{i=1}^n M_i(x_i-x_{i-1}).
\]
\end{definition}

\begin{definition}
The \emph{lower Riemann integral} and \emph{upper Riemann integral} of $f$ over
$[a,b]$ are defined by
\[
\underline{\int_a^b} f
=
\sup_{\mathcal{P}} l(f,\mathcal{P}),
\qquad
\overline{\int_a^b} f
=
\inf_{\mathcal{P}} u(f,\mathcal{P}).
\]
If
\[
\underline{\int_a^b} f
=
\overline{\int_a^b} f,
\]
then $f$ is called \emph{Riemann integrable} on $[a,b]$, and the common value is
denoted by
\[
\int_a^b f(x)\,dx.
\]
\end{definition}

\begin{lemma}[Lemma 2.5.1]
A bounded function $f:[a,b]\to\mathbb{R}$ is Riemann integrable if and only if
for every $\varepsilon>0$ there exists a partition $\mathcal{P}$ of $[a,b]$ such
that
\[
u(f,\mathcal{P})-l(f,\mathcal{P})<\varepsilon.
\]
\end{lemma}

\begin{theorem}
Let $[a,b]$ be a closed bounded interval, and let
$f:[a,b]\to\mathbb{R}$ be bounded. Then:
\begin{enumerate}[label=(\alph*)]
    \item $f$ is Riemann integrable if and only if $f$ is continuous almost
    everywhere on $[a,b]$;
    \item if $f$ is Riemann integrable, then $f$ is Lebesgue integrable and
    \[
    (R)\int_a^b f(x)\,dx
    =
    (L)\int_{[a,b]} f\,d\lambda.
    \]
\end{enumerate}
\end{theorem}

\begin{proof}[Proof sketch]
Assume first that $f$ is Riemann integrable. Choose partitions
$\mathcal{P}_n$ such that
\[
u(f,\mathcal{P}_n)-l(f,\mathcal{P}_n)\to 0,
\]
and refine them if necessary so that $\mathcal{P}_{n+1}$ refines
$\mathcal{P}_n$.

Define simple functions $g_n$ and $h_n$ by letting $g_n$ be the infimum of $f$
on each interval of $\mathcal{P}_n$, and $h_n$ the supremum of $f$ on each such
interval. Then
\[
g_n \le f \le h_n,
\]
with $\{g_n\}$ increasing and $\{h_n\}$ decreasing. Moreover,
\[
\int_{[a,b]} g_n\,d\lambda = l(f,\mathcal{P}_n),
\qquad
\int_{[a,b]} h_n\,d\lambda = u(f,\mathcal{P}_n).
\]
Let
\[
g=\lim_{n\to\infty} g_n,
\qquad
h=\lim_{n\to\infty} h_n.
\]
Since the lower and upper sums have the same limit, we get
\[
\int_{[a,b]} (h-g)\,d\lambda = 0.
\]
Since $h-g\ge 0$, this implies $h=g$ almost everywhere. Outside the partition
points, equality $g(x)=h(x)$ forces $f$ to be continuous at $x$. Hence $f$ is
continuous almost everywhere. Also, since $g\le f\le h$ and $g=h$ almost
everywhere, $f$ agrees almost everywhere with the measurable function $g$.
Therefore $f$ is Lebesgue measurable and integrable, and the Riemann and
Lebesgue integrals agree.

Conversely, suppose that $f$ is continuous almost everywhere. Take partitions
$\mathcal{P}_n$ into equal subintervals and define $g_n,h_n$ as above. Then
\[
h_n-g_n \to 0
\]
almost everywhere. Since $f$ is bounded, the functions $h_n-g_n$ are dominated
by an integrable constant. By the Dominated Convergence Theorem,
\[
\int_{[a,b]} (h_n-g_n)\,d\lambda \to 0.
\]
Thus
\[
u(f,\mathcal{P}_n)-l(f,\mathcal{P}_n)\to 0.
\]
By the Riemann criterion, $f$ is Riemann integrable.
\end{proof}

\begin{proposition}[Triangle inequality for integrals]
Let $(X,\mathcal{A},\mu)$ be a measure space, and let
$f:X\to[-\infty,\infty]$ be measurable. Then $f$ is integrable if and only if
$|f|$ is integrable. In that case,
\[
\left|\int f\,d\mu\right|
\le
\int |f|\,d\mu.
\]
\end{proposition}

\begin{proof}
Recall that
\[
f=f^+-f^-,
\qquad
|f|=f^++f^-.
\]
Thus $f$ is integrable if and only if both $f^+$ and $f^-$ are integrable, which
is equivalent to $|f|$ being integrable. If $f$ is integrable, then
\[
\left|\int f\,d\mu\right|
=
\left|\int f^+\,d\mu-\int f^-\,d\mu\right|
\le
\int f^+\,d\mu+\int f^-\,d\mu
=
\int |f|\,d\mu.
\]
\end{proof}

\begin{proposition}[Proposition 3.1.3]
Let $(X,\mathcal{A},\mu)$ be a measure space, and let $f,f_1,f_2,\dots$ be
real-valued measurable functions on $X$. If $\mu(X)<\infty$ and
\[
f_n \to f
\]
almost everywhere, then $f_n \to f$ in measure.
\end{proposition}

\begin{proof}
We must show that for every $\varepsilon>0$,
\[
\mu\big(\{x\in X: |f_n(x)-f(x)|>\varepsilon\}\big)\to 0.
\]
Fix $\varepsilon>0$ and define
\[
A_n:=\{x\in X: |f_n(x)-f(x)|>\varepsilon\},
\qquad
B_n:=\bigcup_{k=n}^\infty A_k.
\]
Then $\{B_n\}$ is a decreasing sequence of measurable sets. Moreover,
\[
\bigcap_{n=1}^\infty B_n
\subset
\{x\in X : f_n(x) \text{ does not converge to } f(x)\}.
\]
Since $f_n\to f$ almost everywhere, this intersection has measure zero.

Because $\mu(X)<\infty$ and $B_1\subset X$, we have $\mu(B_1)<\infty$. Hence,
by continuity from above of measures, Proposition 1.2.5,
\[
\mu(B_n)\to \mu\!\left(\bigcap_{n=1}^\infty B_n\right)=0.
\]
Since $A_n\subset B_n$, we get
\[
0\le \mu(A_n)\le \mu(B_n)\to 0.
\]
Therefore,
\[
\mu\big(\{x\in X: |f_n(x)-f(x)|>\varepsilon\}\big)\to 0,
\]
so $f_n\to f$ in measure.
\end{proof}

\begin{proposition}
Let $(X,\mathcal{A},\mu)$ be a measure space, and let $f,f_1,f_2,\dots$ be
real-valued measurable functions on $X$. If $f_n\to f$ in measure, then there
exists a subsequence $\{f_{n_k}\}$ such that $f_{n_k}\to f$ almost everywhere.
\end{proposition}

\begin{proof}
Since $f_n\to f$ in measure, for each $k$ we can choose $n_k>n_{k-1}$ such that
\[
\mu\left(\left\{x\in X: |f_{n_k}(x)-f(x)|>\frac1k\right\}\right)
\le \frac1{2^k}.
\]
Define
\[
A_k:=\left\{x\in X: |f_{n_k}(x)-f(x)|>\frac1k\right\}.
\]
Then
\[
\sum_{k=1}^\infty \mu(A_k) \le \sum_{k=1}^\infty \frac1{2^k}<\infty.
\]
Now consider
\[
\limsup_{k\to\infty} A_k
=
\bigcap_{j=1}^\infty \bigcup_{k=j}^\infty A_k.
\]
For every $j$,
\[
\mu\left(\bigcup_{k=j}^\infty A_k\right)
\le
\sum_{k=j}^\infty \mu(A_k)
\le
\sum_{k=j}^\infty \frac1{2^k}
\to 0.
\]
Hence
\[
\mu\left(\limsup_{k\to\infty} A_k\right)=0.
\]

If $x\notin \limsup_{k\to\infty} A_k$, then $x$ belongs to only finitely many
of the sets $A_k$. Therefore, for all sufficiently large $k$,
\[
|f_{n_k}(x)-f(x)|\le \frac1k,
\]
which implies $f_{n_k}(x)\to f(x)$. Thus $f_{n_k}\to f$ almost everywhere.
\end{proof}

\begin{theorem}[Egoroff's Theorem]
Let $(X,\mathcal{A},\mu)$ be a measure space, and let $f,f_1,f_2,\dots$ be
real-valued measurable functions on $X$. If $\mu(X)<\infty$ and
\[
f_n \to f
\]
almost everywhere, then for every $\varepsilon>0$ there exists a set
$B\in\mathcal{A}$ such that
\[
\mu(B^c)<\varepsilon
\]
and $f_n\to f$ uniformly on $B$.
\end{theorem}

\begin{proof}
Fix $\varepsilon>0$. For each $n$, define
\[
g_n:=\sup_{j\ge n}|f_j-f|.
\]
Since $f_n\to f$ almost everywhere, we have
\[
g_n\to 0
\]
almost everywhere. Since $\mu(X)<\infty$, Proposition 3.1.2 implies that
$g_n\to 0$ in measure.

Hence, for each $k$, we can choose $n_k$ such that
\[
\mu\left(\left\{x\in X:g_{n_k}(x)>\frac1k\right\}\right)
<
\frac{\varepsilon}{2^k}.
\]
Define
\[
B_k:=\left\{x\in X:g_{n_k}(x)\le \frac1k\right\},
\qquad
B:=\bigcap_{k=1}^\infty B_k.
\]
Then
\[
B^c=\bigcup_{k=1}^\infty B_k^c,
\]
so by subadditivity,
\[
\mu(B^c)
\le
\sum_{k=1}^\infty \mu(B_k^c)
<
\sum_{k=1}^\infty \frac{\varepsilon}{2^k}
=
\varepsilon.
\]

It remains to show uniform convergence on $B$. Let $\delta>0$ and choose $k$
such that $1/k<\delta$. Since $B\subset B_k$, for every $x\in B$ we have
\[
g_{n_k}(x)\le \frac1k<\delta.
\]
Thus, for every $n\ge n_k$ and every $x\in B$,
\[
|f_n(x)-f(x)|
\le
g_{n_k}(x)
<
\delta.
\]
Therefore $f_n\to f$ uniformly on $B$.
\end{proof}

\begin{proposition}
Let $(X,\mathcal{A},\mu)$ be a measure space, and let
$f,f_1,f_2,\dots \in \mathcal{L}^1(X,\mathcal{A},\mu)$. If $f_n\to f$ in mean, i.e.
\[
\int |f_n-f|\,d\mu \to 0,
\]
then $f_n\to f$ in measure.
\end{proposition}

\begin{proof}
For every $\varepsilon>0$, Proposition 2.3.10 inequality gives
\[
\mu\big(\{x\in X: |f_n(x)-f(x)|>\varepsilon\}\big)
\le
\frac{1}{\varepsilon}\int |f_n-f|\,d\mu.
\]
Since the right-hand side tends to $0$, we get
\[
\mu\big(\{x\in X: |f_n(x)-f(x)|>\varepsilon\}\big)\to 0.
\]
Thus $f_n\to f$ in measure.
\end{proof}

\begin{proposition}[Propositions 3.1.5]
Let $(X,\mathcal{A},\mu)$ be a measure space, and let
$f,f_1,f_2,\dots \in \mathcal{L}^1(X,\mathcal{A},\mu)$. Suppose that $f_n\to f$ almost
everywhere or in measure. If there exists an integrable function
$g:X\to[0,\infty]$ such that
\[
|f_n|\le g
\quad\text{and}\quad
|f|\le g
\]
almost everywhere, then $f_n\to f$ in mean.
\end{proposition}

\begin{proof}
First suppose that $f_n\to f$ almost everywhere. Then
\[
|f_n-f|\to 0
\]
almost everywhere, and
\[
|f_n-f|\le |f_n|+|f|\le 2g
\]
almost everywhere. Since $2g$ is integrable, the Dominated Convergence Theorem
implies
\[
\int |f_n-f|\,d\mu \to 0.
\]
Hence $f_n\to f$ in mean.

Now suppose instead that $f_n\to f$ in measure. By Proposition 3.1.3, every
subsequence of $\{f_n\}$ has a further subsequence that converges to $f$ almost
everywhere. By the first part, every such further subsequence converges to $f$
in mean.

If $f_n$ did not converge to $f$ in mean, then there would exist
$\varepsilon>0$ and a subsequence $\{f_{n_k}\}$ such that
\[
\int |f_{n_k}-f|\,d\mu \ge \varepsilon
\]
for all $k$. But this subsequence has a further subsequence converging to $f$
almost everywhere, and hence in mean by the first part, a contradiction.
Therefore $f_n\to f$ in mean.
\end{proof}

Let $(X,\mathcal{A},\mu)$ be a measure space, and let
$f,f_1,f_2,\dots \in L^1(X,\mathcal{A},\mu)$. Suppose that $f_n\to f$ almost

\begin{definition}
Let $V$ be a real vector space. A function $\|\cdot\|:V\to[0,\infty)$ is called
a \emph{norm} if, for all $x,y\in V$ and $\alpha\in\mathbb{R}$,
\begin{enumerate}[label=(\alph*)]
    \item $\|x\|=0$ if and only if $x=0$,
    \item $\|\alpha x\|=|\alpha|\|x\|$,
    \item $\|x+y\|\le \|x\|+\|y\|$.
\end{enumerate}
A vector space with a norm is called a \emph{normed vector space}.
\end{definition}

\begin{definition}
Let $V$ be a real vector space. A function
$\langle \cdot,\cdot\rangle:V\times V\to\mathbb{R}$ is called an
\emph{inner product} if, for all $x,y,z\in V$ and $\alpha,\beta\in\mathbb{R}$,
\begin{enumerate}[label=(\alph*)]
    \item $\langle x,x\rangle\ge 0$, with equality if and only if $x=0$,
    \item $\langle x,y\rangle=\langle y,x\rangle$,
    \item $\langle \alpha x+\beta y,z\rangle
    =\alpha\langle x,z\rangle+\beta\langle y,z\rangle$.
\end{enumerate}
It induces the norm
\[
\|x\|:=\sqrt{\langle x,x\rangle}.
\]
\end{definition}

\begin{definition}
Let $(X,\mathcal{A},\mu)$ be a measure space and let $1\le p<\infty$. Define
\[
\mathcal{L}^p(X,\mathcal{A},\mu)
:=
\left\{
f:X\to\mathbb{R} \text{ measurable} :
\int |f|^p\,d\mu<\infty
\right\}.
\]
For $f\in\mathcal{L}^p$, define
\[
\|f\|_p
:=
\left(\int |f|^p\,d\mu\right)^{1/p}.
\]
\end{definition}

\begin{definition}
Let
\[
\mathcal{N}^p(X,\mathcal{A},\mu)
:=
\{f\in\mathcal{L}^p(X,\mathcal{A},\mu):\|f\|_p=0\}.
\]
For $1\le p<\infty$, this is precisely the set of functions in
$\mathcal{L}^p$ that vanish almost everywhere.
\end{definition}

\begin{definition}
The space $L^p(X,\mathcal{A},\mu)$ is defined as the quotient space
\[
L^p(X,\mathcal{A},\mu)
:=
\mathcal{L}^p(X,\mathcal{A},\mu)/\mathcal{N}^p(X,\mathcal{A},\mu).
\]
Equivalently, elements of $L^p$ are equivalence classes of functions in
$\mathcal{L}^p$, where
\[
f\sim g
\quad\Longleftrightarrow\quad
f-g\in \mathcal{N}^p
\quad\Longleftrightarrow\quad
f=g \text{ almost everywhere}.
\]
\end{definition}

\begin{remark}
If $\langle f\rangle$ denotes the equivalence class of $f$, then
\[
\langle f\rangle+\langle g\rangle:=\langle f+g\rangle,
\qquad
\alpha\langle f\rangle:=\langle \alpha f\rangle.
\]
Moreover, if $f\sim g$, then $\|f\|_p=\|g\|_p$, so the formula
\[
\|\langle f\rangle\|_p:=\|f\|_p
\]
is well-defined. This gives a norm on $L^p$.
\end{remark}

\begin{definition}
For $p=2$, the space $L^2(X,\mathcal{A},\mu)$ has the inner product
\[
\langle f,g\rangle
:=
\int fg\,d\mu.
\]
The induced norm is
\[
\|f\|_2
=
\left(\int |f|^2\,d\mu\right)^{1/2}.
\]
\end{definition}
